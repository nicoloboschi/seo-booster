Here's a basic Python example demonstrating a simple key-value memory store, which is a foundational concept in AI memory systems:

```python
class SimpleMemory:
 def __init__(self):
 self.memory = {}

 def store(self, key, value):
 """Stores a key-value pair in memory."""
 self.memory[key] = value
 print(f"Stored: '{key}' -> '{value}'")

 def recall(self, key):
 """Recalls a value associated with a key."""
 return self.memory.get(key, "Information not found.")

 def forget(self, key):
 """Removes a key-value pair from memory."""
 if key in self.memory:
 del self.memory[key]
 print(f"Forgot: '{key}'")
 else:
 print(f"Cannot forget: '{key}' not found.")

## Example Usage
agent_memory = SimpleMemory()
agent_memory.store("user_preference", "likes_python")
agent_memory.store("last_task", "writing_code")

print(f"Recall user_preference: {agent_memory.recall('user_preference')}")
print(f"Recall last_task: {agent_memory.recall('last_task')}")
print(f"Recall non_existent_key: {agent_memory.recall('non_existent_key')}")

agent_memory.forget("last_task")
print(f"Recall last_task after forgetting: {agent_memory.recall('last_task')}")
```

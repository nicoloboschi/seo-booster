import numpy as np
from sklearn.metrics.pairwise import cosine_similarity
from typing import List, Dict, Tuple

class VectorMemorySystem:
 def __init__(self, embedding_dim: int = 128):
 self.embedding_dim = embedding_dim
 # Stores memories as tuples: (id, text, embedding_vector)
 self.memory_store: List[Tuple[int, str, np.ndarray]] = []
 self.next_id = 0
 print("Vector memory system initialized.")

 def add_memory(self, text: str, embedding_vector: np.ndarray):
 """Adds a memory with its text and pre-computed embedding vector."""
 if embedding_vector.shape[0] != self.embedding_dim:
 raise ValueError(f"Embedding dimension mismatch. Expected {self.embedding_dim}, got {embedding_vector.shape[0]}.")

 memory_id = self.next_id
 self.memory_store.append((memory_id, text, embedding_vector))
 self.next_id += 1
 print(f"Memory added with ID {memory_id}: '{text[:50]}...'")

 def search_memory(self, query_embedding: np.ndarray, top_k: int = 3) -> List[Tuple[int, str, float]]:
 """Searches for semantically similar memories using cosine similarity."""
 if not self.memory_store:
 print("Memory store is empty. No search results.")
 return []

 if query_embedding.shape[0] != self.embedding_dim:
 raise ValueError(f"Query embedding dimension mismatch. Expected {self.embedding_dim}, got {query_embedding.shape[0]}.")

 # Extract just the embedding vectors for efficient calculation
 memory_embeddings = np.array([mem[2] for mem in self.memory_store])

 # Calculate cosine similarity between query and all memory embeddings
 similarities = cosine_similarity(query_embedding.reshape(1, -1), memory_embeddings)[0]

 # Get indices of top_k most similar memories
 # argsort sorts in ascending order, so we take the last top_k elements and reverse them
 top_indices = np.argsort(similarities)[-top_k:][::-1]

 results = []
 for i in top_indices:
 mem_id, mem_text, _ = self.memory_store[i]
 similarity_score = similarities[i]
 # Only include results with a positive similarity score
 if similarity_score > 0:
 results.append((mem_id, mem_text, similarity_score))

 print(f"Found {len(results)} similar memories for the query.")
 return results

 def get_memory_by_id(self, memory_id: int) -> Tuple[str, np.ndarray] | None:
 """Retrieves a specific memory by its ID."""
 for mem_id, mem_text, mem_vector in self.memory_store:
 if mem_id == memory_id:
 return mem_text, mem_vector
 return None

class SimpleAgentMemory:
 def __init__(self):
 self.memory = {}
 self.event_log = [] # To store a sequence of events for episodic memory

 def store_fact(self, key, value):
 """Stores a fact in the agent's memory."""
 self.memory[key] = value
 print(f"Stored fact: {key} -> {value}")

 def retrieve_fact(self, key):
 """Retrieves a fact from the agent's memory."""
 return self.memory.get(key, "Fact not found.")

 def log_event(self, event_description):
 """Logs an event, contributing to episodic memory."""
 self.event_log.append(event_description)
 print(f"Event logged: {event_description}")

 def recall_last_n_events(self, n):
 """Recalls the last n logged events."""
 if not self.event_log:
 return "No events logged yet."
 return self.event_log[-n:]
